local function fit(s, width)
    s = tostring(s or "")
    if width <= 0 then return "" end
    local ok, w = pcall(function() return unicode.wlen(s) end)
    if ok and w and w > width then
        local ok2, r = pcall(function() return unicode.wtrunc(s, width) end)
        if ok2 and r then return r end
        return s:sub(1, width)
    elseif not ok and #s > width then
        return s:sub(1, width)
    end
    return s
end
local function ulen(s)
    s = tostring(s or "")
    local ok, n = pcall(function() return unicode.wlen(s) end)
    return (ok and n) or #s
end
local function usub(str, a, b)
    str = tostring(str or "")
    local ok, r = pcall(function() return unicode.sub(str, a, b) end)
    return (ok and r) or str:sub(a, b)
end
local function wrapText(text, width, maxLines)
    text = tostring(text or "")
    local out, cur = {}, ""
    if width <= 0 then return out end
    local n = ulen(text)
    for i2 = 1, n do
        local ch = usub(text, i2, i2)
        if ch == "\n" then
            out[#out + 1] = cur
            cur = ""
        else
            local try = cur .. ch
            if ulen(try) > width then
                out[#out + 1] = cur
                cur = ch
            else
                cur = try
            end
        end
        if maxLines and #out >= maxLines then return out end
    end
    if cur ~= "" and (not maxLines or #out < maxLines) then out[#out + 1] = cur end
    return out
end
local function waitInput()
    while true do
        local e = { event.pull() }     -- 不带过滤器、不带超时：纯阻塞，最稳
        local name = e[1]
        if name == "key_down" then
            return "key_down", e[3], e[4]
        elseif name == "touch" then
            return "touch", e[3], e[4]
        end
    end
end
local function padRight(s, width)
    s = tostring(s or "")
    local ok, w = pcall(function() return unicode.wlen(s) end)
    w = (ok and w) or #s
    local pad = width - w
    if pad < 1 then pad = 1 end
    return s .. string.rep(" ", pad)
end

local function materialFromLabel(lb)
    lb = tostring(lb or "")
    if lb:find("[Ii]ngot$") then
        local mat = lb:gsub("^Hot%s+", "")
        mat = mat:gsub("%s*[Ii]ngot$", "")
        mat = mat:lower()
        mat = mat:gsub("%s+", "")
        if mat == "" then return nil end
        return mat
    end
    if lb:sub(-3) == "锭" then
        local mat = lb:gsub("^热", "")
        mat = mat:gsub("锭$", "")

    end
    return nil

local function ingotMaterial(it, labels)
    local lb0 = tostring(it.label or "")
    if lb0 ~= "" then
        local m0 = materialFromLabel(lb0)
        if m0 then return m0 end
    end
    if labels then
        local key = string.format("%s:%d", tostring(it.name or ""), tonumber(it.damage) or 0)
        local lb = labels[key]
        if lb and lb ~= "" then
            local m1 = materialFromLabel(lb)
            if m1 then return m1 end

    end
    local od = tostring(it.oreDict or "")
    if od ~= "" then

local function convertOutputs(outs, fluidMap, manual, known, labels)
    local newOuts, log, skipped = {}, {}, {}
    for _, e in ipairs(outs) do
        if e.fluid then
            newOuts[#newOuts + 1] = e
            log[#log + 1] = { kind = "keepFluid", from = e, to = e }
        elseif isIngot(e) then
            local mat = ingotMaterial(e, labels)
            local fname = (manual and manual[entryKey(e)])
                or (known and known[entryKey(e)])
                or (mat and fluidMap[mat]) or nil
            if not fname and mat and not fluidMap[mat] then
                fname = FLUID_PREFIX .. mat
            end
            if fname then
                local n = (tonumber(e.size) or 1) * MB_PER_INGOT
                local ne = {
                    name = fname,
                    damage = 0,
                    size = n,
                    fluid = true,
                    label = "",
                    hasTag = false,
                }
                newOuts[#newOuts + 1] = ne
                log[#log + 1] = { kind = "converted", from = e, to = ne, mat = mat }
            else
                newOuts[#newOuts + 1] = e
                skipped[#skipped + 1] = e
                log[#log + 1] = { kind = "fail", from = e, to = e }
            end
        else
            newOuts[#newOuts + 1] = e
            log[#log + 1] = { kind = "keepOther", from = e, to = e }
        end
    end
    return newOuts, log, skipped
end
local function dumpEntry(it)
    if type(it) ~= "table" then return tostring(it) end
    local ks = {}
    for k in pairs(it) do ks[#ks + 1] = tostring(k) end
    table.sort(ks)
    local parts = {}
    for _, k in ipairs(ks) do
        local v = it[k]
        local vs
        if type(v) == "table" then vs = "<table>" else vs = tostring(v) end
        parts[#parts + 1] = k .. "=" .. vs
    end
    return table.concat(parts, " ")
end
local function writeOne(iface, kind, slot, idx, e)
    local candidates
    if e.fluid then
        local short = e.name
        local long = e.name
        if short:sub(1, 9) == "gregtech:" then
            short = short:sub(10)
        else
            long = "gregtech:" .. short
        end
        candidates = {
            { data = { name = short, size = e.size }, tag = "fluid", label = "size/短名" },
            { data = { name = long, size = e.size }, tag = "fluid", label = "size/长名" },
            { data = { name = short, amount = e.size }, tag = "fluid", label = "amount/短名" },
            { data = { name = long, amount = e.size }, tag = "fluid", label = "amount/长名" },
        }
    else
        local d = { name = e.name, size = e.size }
        if e.damage and e.damage ~= 0 then d.damage = e.damage end
        if e.tag then d.tag = e.tag end
        candidates = { { data = d, tag = "item", label = "item" } }
    end
    local lastErr = nil
    for _, c in ipairs(candidates) do
        pcall(function()
            if kind == "out" then
                return iface.setInterfacePatternOutput(slot, idx, c.data, c.tag)
            else
                return iface.setInterfacePatternInput(slot, idx, c.data, c.tag)
            end
        end)
        os.sleep(0.15)
        local okR, pat = pcall(iface.getInterfacePattern, slot)
        if okR and pat then
            local list = (kind == "out") and pat.outputs or pat.inputs
            local got = list and list[idx]
            local gotName = got and tostring(got.name) or nil
            if gotName == e.name then
                return true, c.label
            end
            if gotName == tostring(c.data.name) then
                return true, c.label
            end
            if gotName == nil then
                return true, c.label .. "(未回读)"
            end
            lastErr = string.format("%s: 回读是 %s", c.label, gotName)
        else
            return true, c.label .. "(未回读)"
        end
    end
    return false, lastErr or "四种写法都试过了"
end
local function writePattern(iface, slot, outs, ins, dry)
    local notes = {}
    local function put(kind, list)
        for idx, e in ipairs(list) do
            if dry then
                notes[#notes + 1] = string.format("%s#%d 跳过(预览)", kind, idx)
            else
                local ok, how = writeOne(iface, kind, slot, idx, e)
                if not ok then
                    return false, string.format("%s 第 %d 项失败：%s", kind, idx, tostring(how))
                end
                notes[#notes + 1] = string.format("%s#%d=%s", kind, idx, tostring(how))
            end
            os.sleep(0)
        end
        return true
    end
    local ok1, err1 = put("in", ins)
    if not ok1 then return false, err1 end
    local ok2, err2 = put("out", outs)
    if not ok2 then return false, err2 end
    if not dry then
        local chk = iface.getInterfacePattern(slot)
        if chk then
            for kind, want in pairs({ ["in"] = #ins, ["out"] = #outs }) do
                local list = (kind == "in") and chk.inputs or chk.outputs
                local n = 0
                if list then for _ in pairs(list) do n = n + 1 end end
                if n > want then
                    notes[#notes + 1] = string.format("警告：%s 还有 %d 项多余的老条目（%d 应为 %d）",
                        kind, n - want, n, want)
                end
            end
        end
    end
    return true, table.concat(notes, " "), true
end
local function fmtEntry(e)
    return string.format("%s ×%s", brief(e, 10), numStr(e.size))
end

local function materialOf(label)
    local m = label
    m = m:gsub("^Tiny%s+", ""):gsub("^Small%s+", ""):gsub("^Large%s+", "")
    m = m:gsub("^Huge%s+", ""):gsub("^Medium%s+", ""):gsub("^Long%s+", "")
    m = m:gsub("^Hot%s+", "")
    for _, w in ipairs(SHAPE_WORDS) do
        m = m:gsub("%s*" .. w .. "$", "")
    end
    m = m:gsub("^热", "")
    for _, w in ipairs(CN_SHAPES) do
        if #m >= #w and m:sub(-#w) == w then
            m = m:sub(1, #m - #w)
            break
        end
    end
    m = m:gsub("^Block%s+of%s+", "")
    m = m:gsub("^Block%s+", "")
    m = m:gsub("^of%s+", "")
    m = m:gsub("%s*Fluid$", "")
    m = m:gsub("^%s+", ""):gsub("%s+$", "")
    if m == "" then return nil, nil end
    return CN_MATERIAL[m] or m, m
end

local function fluidCandidates(mat)

        return out
    end
    local key = mat:lower():gsub("[^%w]", "")
    if ALIASES[key] then
        out[#out + 1] = ALIASES[key]
    end
    local a = mat:lower():gsub("%s+", "_")
    local b = mat:lower():gsub("%s+", "")
    out[#out + 1] = "molten." .. a
    if b ~= a then out[#out + 1] = "molten." .. b end
    return out
end

local function usedSlots(ifaces)
    local out = {}
    for _, iface in ipairs(ifaces) do
        for slot = 1, 36 do
            local ok, p = pcall(iface.getInterfacePattern, slot)
            if ok and p then
                out[#out + 1] = { iface = iface, slot = slot }
            end
        end
    end
    return out
end

local function clearDbs(dbs)
    for _, db in ipairs(dbs) do
        local size = 64
        local ok, n = pcall(db.size)
        if ok and type(n) == "number" and n > 0 then
            size = n
        end
        for s = 1, size do
            pcall(db.clear, s)
        end
        os.sleep(0)
    end
end
local function buildDbCache(dbs)
    local cache, empties = {}, {}
    for _, db in ipairs(dbs) do
        local size = 64
        local ok, n = pcall(db.size)
        if ok and type(n) == "number" and n > 0 then
            size = n
        end
        for s = 1, size do
            local okG, v = pcall(db.get, s)
            if okG then
                local vname = nil
                local vdmg = 0
                if type(v) == "table" then
                    vname = v.name
                    vdmg = tonumber(v.damage) or 0
                end
                if vname and vname ~= "" then
                    local key = tostring(vname) .. "|" .. tostring(vdmg)
                    if not cache[key] then
                        cache[key] = { db = db, slot = s }
                    end
                else
                    empties[#empties + 1] = { db = db, slot = s }
                end
            end
        end
        os.sleep(0)
    end
    return cache, empties
end
local function dbSlotFor(cache, empties, name, damage, snbt)
    local key = name .. "|" .. tostring(damage)
    local hit = cache[key]
    if hit then
        return hit.db, hit.slot
    end
    local e = table.remove(empties, 1)
    if e then
        local okS = pcall(e.db.set, e.slot, name, damage, snbt)
        if okS then
            cache[key] = { db = e.db, slot = e.slot }
            return e.db, e.slot
        end
        table.insert(empties, 1, e)
    end
    return nil, nil
end
local function writeOne(s, w, dbs, cache, empties)
    local nErr = nil
    local iface, slot = s.iface, s.slot
    for i, o in ipairs(w.outs) do
        local ok, err
        if o.rename and #dbs > 0 then
            local snbt = '{display:{Name:"' .. o.rename .. '"}}'
            local rdb, ds = dbSlotFor(cache, empties, o.name, o.damage or 0, snbt)
            if rdb then
                ok, err = pcall(iface.setInterfacePatternOutput, slot, i, rdb.address, ds, o.size or 1)
            else
                ok, err = false, "数据库槽不够"
            end
        elseif o.fluid then
            ok, err = pcall(iface.setInterfacePatternOutput, slot, i, { name = o.name, size = o.size }, "fluid")
        else
            ok, err = pcall(iface.setInterfacePatternOutput, slot, i,
                { name = o.name, size = o.size, damage = o.damage or 0 }, "item")
        end
        if not ok then nErr = tostring(err) end
    end
    for i, e in ipairs(w.ins) do
        local ok, err
        if e.rename and e.renameTo and #dbs > 0 then
            local snbt = '{display:{Name:"' .. e.renameTo .. '"}}'
            local rdb, ds = dbSlotFor(cache, empties, e.name, e.damage or 0, snbt)
            if rdb then
                ok, err = pcall(iface.setInterfacePatternInput, slot, i, rdb.address, ds, e.size)
            else
                ok, err = false, "数据库槽不够"
            end
        elseif e.fluid then
            ok, err = pcall(iface.setInterfacePatternInput, slot, i, { name = e.name, size = e.size }, "fluid")
        else
            ok, err = pcall(iface.setInterfacePatternInput, slot, i,
                { name = e.name, size = e.size, damage = e.damage or 0 }, "item")
        end
        if not ok then nErr = tostring(err) end
    end
    return nErr
end

local function genNAC(idx, renameTo)
    local r = RECIPES[idx]
    local ins = {}
    for _, e in ipairs(r.ins) do
        if e.rename and (e.size or 0) > 64 then
            local left = e.size
            while left > 0 do
                local n = math.min(left, 64)
                local t = item(e.name, n, e.damage)
                t.fluid = e.fluid
                t.rename = true
                t.renameTo = renameTo or e.label
                ins[#ins + 1] = t
                left = left - n
            end
        else
            local t = item(e.name, e.size, e.damage)
            t.fluid = e.fluid
            if e.rename then
                t.rename = true
                t.renameTo = renameTo or e.label
            end
            ins[#ins + 1] = t
        end
    end
    local outs = {}
    for _, o in ipairs(r.out) do
        outs[#outs + 1] = item(o.name, o.size, o.damage)
    end
    return { { tag = r.name, outs = outs, ins = ins } }
end

local function splitName(rest)
    local best = nil
    for _, pat in ipairs({ " damage=", " size=", " amount=", " label=" }) do
        local p = rest:find(pat, 1, true)
        if p and (not best or p < best) then
            best = p
        end
    end
    if best then
        local name = rest:sub(1, best - 1):gsub("^%s+", ""):gsub("%s+$", "")
        return name, rest:sub(best + 1)
    end
    return rest:match("^(%S+)") or "", rest
end
local function parseLine(line)
    local ifaceN, slot, dir, idx, rest = line:match("^%[接口(%d+) 槽(%d+)%]%s+(%a+)%s+(%d+)%s+(.+)$")
    if not dir then
        dir, idx, rest = line:match("^%s*(%a+)%s+(%d+)%s+(.+)$")
        ifaceN, slot = 1, 1
    end
    if dir ~= "IN" and dir ~= "OUT" then
        return nil
    end
    local e = {}
    e.name, rest = splitName(rest)
    if not e.name or e.name == "" then
        return nil
    end
    e.damage = tonumber(rest:match("damage=([%-%d]+)")) or 0
    e.size = tonumber(rest:match("size=([%-%d]+)"))
        or tonumber(rest:match("amount=([%-%d]+)")) or 1
    e.label = rest:match('label="([^"]*)"')
    return tonumber(ifaceN) or 1, tonumber(slot) or 1, dir, tonumber(idx), e
end
local function parseText(text)
    local patterns = {}
    for line in text:gmatch("[^\r\n]+") do
        local ifaceN, slot, dir, idx, e = parseLine(line)
        if ifaceN and slot and idx then
            local bySlot = patterns[ifaceN]
            if not bySlot then
                bySlot = {}
                patterns[ifaceN] = bySlot
            end
            local p = bySlot[slot]
            if not p then
                p = { ins = {}, outs = {} }
                bySlot[slot] = p
            end
            if dir == "IN" then
                p.ins[idx] = e
            else
                p.outs[idx] = e
            end
        end
    end
    return patterns
end
local function markRenames(patterns, dbs)
    if #dbs == 0 then return end
    local db = dbs[1]
    local defaultLabel = {}
    local slot = 0
    local function defaultOf(name, damage)
        local key = tostring(name) .. "|" .. tostring(damage)
        if defaultLabel[key] ~= nil then
            return defaultLabel[key]
        end
        slot = slot % 64 + 1
        local ok = pcall(db.set, slot, name, damage, "")
        if not ok then
            defaultLabel[key] = ""
            return ""
        end
        local ok2, v = pcall(db.get, slot)
        if not ok2 or type(v) ~= "table" then
            defaultLabel[key] = ""
            return ""
        end
        defaultLabel[key] = tostring(v.label or "")
        return defaultLabel[key]
    end
    local function mark(list)
        for _, e in pairs(list or {}) do
            local def = defaultOf(e.name, e.damage)
            e.rename = (def ~= "" and (e.label or "") ~= def)
        end
    end
    for _, bySlot in pairs(patterns) do
        for _, p in pairs(bySlot) do
            mark(p.ins)
            mark(p.outs)
        end
    end
end

    local nPat = 0
    local maxIface = 0
    for ifaceN, bySlot in pairs(patterns) do
        for _ in pairs(bySlot) do nPat = nPat + 1 end
        if ifaceN > maxIface then maxIface = ifaceN end
    end
    if nPat == 0 then

            local ok, res = pcall(fn, ...)

                return false

            return res and true or false

                    end
                end

            end
        end
    end

end

local function endsWith(s, suffix)
    s = tostring(s or "")
    return #s >= #suffix and s:sub(-#suffix) == suffix
end
local function labelOf(e)
    return tostring(e.label or "")
end
local function listPatterns(iface)
    local out = {}
    for slot = 1, 36 do
        local ok, p = pcall(iface.getInterfacePattern, slot)
        if ok and p then
            out[#out + 1] = { slot = slot, pat = p, iface = iface }
        end
        os.sleep(0)
    end
    return out
end
local function learnPlates(patterns)
    local plates = {}
    local function scan(list)
        for _, e in pairs(list or {}) do
            local lb = labelOf(e)
            if endsWith(lb, "板") then
                local mat = lb:sub(1, #lb - #"板")
                if mat ~= "" and not plates[mat] then
                    plates[mat] = { name = e.name, damage = tonumber(e.damage) or 0 }
                end
            end
        end
    end
    for _, item in ipairs(patterns) do
        scan(item.pat.inputs)
        scan(item.pat.outputs)
    end
    return plates
end
local function guessPlate(e)
    local name = tostring(e.name or "")
    local dmg = tonumber(e.damage) or 0
    local plateName = name:gsub("[Ii]ngot", "plate")
    if plateName ~= name then
        return { name = plateName, damage = dmg }
    end
    if name == "gregtech:gt.metaitem.01" and dmg >= 11000 and dmg < 12000 then
        return { name = name, damage = dmg + 6000 }
    end
    if dmg >= 11000 and dmg < 12000 then
        return { name = name, damage = dmg + 6000 }
    end
    return nil
end

                hasFoil = true
                break
            end
        end

            local changed = 0
            for idx, e in pairs(pat.inputs or {}) do
                local lb = labelOf(e)

                    local mat = lb:sub(1, #lb - #"锭")

                        end
                    else
                        problems[#problems + 1] = string.format("槽 %d 认不出「%s」对应的板", item.slot, mat)
                    end
                end
            end
            if changed > 0 then
                nDone = nDone + 1
            else
                nSkip = nSkip + 1
            end
        end
        os.sleep(0)
    end
    ctx.write(string.rep("-", 56))
    ctx.write(string.format("完成：改 %d 个样板 / 跳过 %d / 失败 %d", nDone, nSkip, nFail))
    if ctx.select then
        local items = {}
        for _, msg in ipairs(problems) do
            items[#items + 1] = { label = msg, desc = "没改的原因", value = 0 }
        end
        items[#items + 1] = { label = "继续", desc = "重新扫描再跑一遍", value = 1 }

            footer = "只处理输出含「箔」的样板",
        })
        if again == 1 then
            return M.run(ctx, args)
        end
    end
    return nFail == 0
end

local STAMP = TOOL_DIR .. "/.installed"
local function embedFingerprint()
    local n = 0
    for path, body in pairs(EMBEDDED) do
        n = n + #path + #body
    end
    return tostring(n)
end
local function bootstrapTools()
    local fp = embedFingerprint()
    local old = ""
    local okR = pcall(function()
        local fh = io.open(STAMP, "r")
        if not fh then return end
        old = fh:read("*a") or ""
        fh:close()
    end)
    if not okR then old = "" end
    old = old:gsub("%s+$", "")
    local missing = 0
    for path in pairs(EMBEDDED) do
        if not fs.exists(path) then missing = missing + 1 end
    end
    if missing == 0 and old == fp then
        return false, 0
    end
    if not fs.isDirectory(TOOL_DIR) then
        pcall(function() fs.makeDirectory(TOOL_DIR) end)
    end
    if fs.isDirectory(TOOL_DIR) then
        local okL, names = pcall(function() return fs.list(TOOL_DIR) end)
        if okL and type(names) == "table" then
            for _, nm in ipairs(names) do
                if type(nm) == "string" and nm:sub(-4) == ".lua" then
                    local oldPath = TOOL_DIR .. "/" .. nm
                    if not EMBEDDED[oldPath] then
                        pcall(function() fs.remove(oldPath) end)
                    end
                end
            end
        end
    end
    local n = 0
    for path, body in pairs(EMBEDDED) do
        local ok, f = pcall(io.open, path, "w")
        if ok and f then
            f:write(body)
            f:close()
            local chk = io.open(path, "r")
            local back = chk and chk:read("*a") or ""
            if chk then chk:close() end
            if back == body then n = n + 1 end
        end
    end
    local ok2, sf = pcall(io.open, STAMP, "w")
    if ok2 and sf then
        sf:write(fp)
        sf:close()
    end
    return n > 0, n
end
local function loadTools()
    local list, bad = {}, {}
    if not fs.isDirectory(TOOL_DIR) then
        return list, bad, "目录不存在：" .. TOOL_DIR
    end
    for file in fs.list(TOOL_DIR) do
        local name = tostring(file)
        if name ~= "." and name ~= ".." and name:sub(1, 1) ~= "_" and name:sub(-4) == ".lua" then
            local path = TOOL_DIR .. "/" .. name
            local ok, mod = pcall(function()
                local chunk = loadfile(path)
                if not chunk then return nil end
                return chunk()
            end)
            if ok and type(mod) == "table" and type(mod.run) == "function" then
                mod._file = name
                mod.id = mod.id or name:sub(1, -5)
                mod.name = mod.name or mod.id
                mod.group = mod.group or "其他"
                mod.order = tonumber(mod.order) or 100
                list[#list + 1] = mod
            else
                bad[#bad + 1] = { file = name, err = tostring(mod) }
            end
        end
    end
    table.sort(list, function(a, b)
        if a.group ~= b.group then return a.group < b.group end
        if a.order ~= b.order then return a.order < b.order end
        return a.name < b.name
    end)
    return list, bad
end

    HITS[#HITS + 1] = { x1 = x, x2 = x + ulen(text) - 1, y = y, action = action }

local function hitAt(x, y)
    for _, h in ipairs(HITS) do
        local y1 = h.y1 or h.y
        local y2 = h.y2 or h.y
        if y >= y1 and y <= y2 and x >= h.x1 and x <= h.x2 then
            return h.action
        end
    end
    return nil
end
local function drawCard(x, y, w, h, t, isSel)
    local bg = isSel and "selected" or "surface"
    local bd = isSel and "borderOn" or "border"
    useBg(bg)
    useFg("text")
    for r = 0, h - 1 do
        term.setCursor(x, y + r)
        term.write(string.rep(" ", w))
    end
    useFg(bd)
    useBg(bg)
    term.setCursor(x, y)
    term.write("┌" .. string.rep("─", w - 2) .. "┐")
    for r = 1, h - 2 do
        term.setCursor(x, y + r); term.write("│")
        term.setCursor(x + w - 1, y + r); term.write("│")
    end
    term.setCursor(x, y + h - 1)
    term.write("└" .. string.rep("─", w - 2) .. "┘")
    local inner = w - 4
    put(x + 2, y + 1, t.name, isSel and "accent" or "text", bg)
    local descLines = wrapText(t.desc or "", inner, h - 3)
    for k, ln in ipairs(descLines) do
        put(x + 2, y + 1 + k, ln, "muted", bg)
    end

local function makeCtx(opts)

                if startIdx > #items - perPage + 1 then startIdx = #items - perPage + 1 end
            end

            local y = top
            for i2 = startIdx, lastIdx do

                local isSel = (i2 == sel)

                useBg(bg)
                pcall(function() term.clearLine() end)
                local mark = isSel and "> " or "  "

                foot2 = string.format("[%d-%d/%d] ", startIdx, lastIdx, #items) .. foot
            end
            put(ulen(btxt) + 2, H, foot2, "muted", "canvas")

                local act = hitAt(a, b)
                if act and act.kind == "pick" then
                    local it = items[act.idx]
                    if it then return it.value, it end
                elseif act and act.kind == "back" then
                    return nil
                end
            elseif ev == "key_down" then
                local ch, code = a, b
                if code == 200 then
                    sel = math.max(1, sel - 1)
                elseif code == 208 then
                    sel = math.min(#items, sel + 1)
                elseif code == 28 then
                    local it = items[sel]
                    if it then return it.value, it end
                elseif ch == 113 or ch == 81 then
                    return nil
                elseif ch and ch >= 49 and ch <= 57 then
                    local n2 = tonumber(string.char(ch))
                    if n2 and items[n2] then return items[n2].value, items[n2] end
                end

        end

local UPDATE_URL = "https://gist.githubusercontent.com/kAGuyAk/45f36eeca31fb2fc536a000d5d9d3a80/raw/gistfile1.txt"
local function doUpdate()
    local addr
    for a in component.list("internet") do
        addr = a
        break
    end
    if not addr then
        return nil, "需要因特网卡"
    end
    local net = component.proxy(addr)
    local ok, data = pcall(function()
        local url = UPDATE_URL .. "?_=" .. tostring(os.time())
        local handle = net.request(url)
        if not handle then return nil end
        local chunks = {}
        while true do
            local chunk = handle.read(8192)
            if not chunk then break end
            chunks[#chunks + 1] = chunk
        end
        pcall(function() handle.close() end)
        return table.concat(chunks)
    end)
    if not ok or type(data) ~= "string" or #data < 10000 then
        return nil, "下载失败（检查网络）"
    end
    local newVer = data:match('local VERSION = "([^"]+)"')
    if not newVer then
        return nil, "下载内容不是工具箱"
    end
    if newVer == VERSION then
        return newVer, "已是最新"
    end
    local path = "/home/toolbox"
    local tmp = path .. ".new"
    local f = io.open(tmp, "w")
    if not f then
        return nil, "写文件失败"
    end
    f:write(data)
    f:close()
    local rf = io.open(tmp, "r")
    if not rf then
        return nil, "写文件失败"
    end
    local check = rf:read(200)
    rf:close()
    if not check or not check:find("local VERSION") then
        os.remove(tmp)
        return nil, "写入校验失败"
    end
    os.remove(path)
    local rok = os.rename(tmp, path)
    if not rok then
        return nil, "替换失败"
    end
    return newVer, nil
end

    put(1 + ulen("  工具箱") + 1, 1, "v" .. VERSION, "muted", "canvas")
    local uptxt = "[更新]"
    local upx = W - ulen(total) - ulen(uptxt) - 2
    put(upx, 1, uptxt, "accent", "surface")
    HITS[#HITS + 1] = { x1 = upx, x2 = upx + ulen(uptxt) - 1, y1 = 1, y2 = 1,
                        action = { kind = "update" } }
    put(W - ulen(total), 1, total, "muted", "canvas")
    rule(2)
    local top = 4
    local bottom = H - 3
    local gapX = 2
    local cw = math.floor((W - gapX * (COLS + 1)) / COLS)
    if cw < 16 then cw = 16 end
    local stepY = CARD_H + 1
    local rowsVisible = math.max(1, math.floor((bottom - top + 1) / stepY))
    if #tools == 0 then
        put(2, top, "一个工具都没有", "warn", "canvas")
        put(2, top + 1, "把 .lua 工具文件放进 " .. TOOL_DIR .. "/", "muted", "canvas")
        put(2, top + 2, "或者删掉主程序重新下载（它会自动补齐）", "muted", "canvas")
    else
        for i, t in ipairs(tools) do
            local slot = i - 1 - scroll * COLS
            local row = math.floor(slot / COLS)
            local col = slot % COLS
            if row >= 0 and row < rowsVisible then
                local x = 1 + gapX + col * (cw + gapX)
                local y = top + row * stepY
                drawCard(x, y, cw, CARD_H, t, i == sel)
                HITS[#HITS + 1] = {
                    x1 = x, x2 = x + cw - 1,
                    y1 = y, y2 = y + CARD_H - 1,
                    action = { kind = "tool", idx = i },
                }
            end
        end
    end
    if #bad and #bad > 0 then
        put(1, H - 2, fit(string.format("  !! %d 个工具加载失败：%s", #bad, bad[1].file), W),
            "error", "canvas")
    elseif status and status ~= "" then
        put(1, H - 2, "  " .. status, "warn", "canvas")
    else
        rule(H - 2)
    end
    local bx = 1
    local function button(label, act, fgRole)
        local txt = "[" .. label .. "]"
        put(bx, H - 1, txt, fgRole or "accent", "surface")
        HITS[#HITS + 1] = { x1 = bx, x2 = bx + ulen(txt) - 1, y1 = H - 1, y2 = H - 1, action = act }
        bx = bx + ulen(txt) + 1
    end

    button("重扫", { kind = "rescan" })
    button("退出", { kind = "quit" }, "muted")
    local author = "作者：柳叶刀kk，任何反馈或期望的功能可在NH社区群at哈基米"
    local aw = ulen(author)
    if aw > W - 1 then
        author = fit(author, W - 1)
        aw = ulen(author)
    end
    put(W - aw + 1, H - 1, author, "muted", "canvas")
    term.setCursor(1, H - 1)
end
local function menuHeight(tools)
    local h2, lastG = 0, nil
    for i3 = 1, #tools do
        if tools[i3].group ~= lastG then
            lastG = tools[i3].group
            h2 = h2 + 1
        end
        h2 = h2 + 1
    end
    return h2
end
local function runMenu(tools, bad)
    local sel, scroll = 1, 0
    local status = ""
    local function runOne(t)
        term.setCursorBlink(false)
        term.clear()
        term.setCursor(1, 1)
        local ctx = makeCtx()
        local ok, err = pcall(t.run, ctx, {})
        if not ok then
            term.setCursor(1, H)
            term.write(fit("!! 工具出错：" .. tostring(err), W))
            term.setCursor(1, H)
            term.write(fit("[按任意键返回菜单]", W))
            waitInput()
        elseif not ctx.hadSelect then
            term.setCursor(1, H)
            term.write(fit("[按任意键返回菜单]", W))
            waitInput()
        end
    end
    while true do
        local top, bottom = 4, H - 3
        local stepY = CARD_H + 1
        local rowsVisible = math.max(1, math.floor((bottom - top + 1) / stepY))
        local selRow = math.floor((sel - 1) / COLS)
        if selRow < scroll then scroll = selRow end
        if selRow >= scroll + rowsVisible then scroll = selRow - rowsVisible + 1 end
        if scroll < 0 then scroll = 0 end
        drawMenu(tools, bad, sel, scroll, status)
        status = ""
        local ev, a, b = waitInput()
        if ev == "touch" then
            local act = hitAt(a, b)
            if act then
                if act.kind == "tool" then
                    if act.idx == sel then
                        local t = tools[sel]
                        if t then runOne(t) end
                    else
                        sel = act.idx
                    end
                elseif act.kind == "run" then
                    local t = tools[sel]
                    if t then runOne(t) end
                elseif act.kind == "rescan" then
                    tools, bad = loadTools()
                    if sel > #tools then sel = math.max(1, #tools) end
                    status = string.format("已重扫：%d 个工具", #tools)
                elseif act.kind == "update" then
                    local ver, msg = doUpdate()
                    if ver then
                        if msg == "已是最新" then
                            status = "已是最新：" .. ver
                        else
                            status = "已更新到 " .. ver .. "，输入 toolbox 重新运行"
                        end
                    else
                        status = "更新失败：" .. tostring(msg)
                    end
                elseif act.kind == "quit" then
                    term.clear()
                    term.setCursor(1, 1)
                    return
                end
            end
        elseif ev == "key_down" then
            local ch, code = a, b
            if code == 200 then
                if sel - COLS >= 1 then sel = sel - COLS end
            elseif code == 208 then
                if sel + COLS <= #tools then sel = sel + COLS end
            elseif code == 203 then
                if (sel - 1) % COLS > 0 then sel = sel - 1 end
            elseif code == 205 then
                if (sel - 1) % COLS < COLS - 1 and sel < #tools then sel = sel + 1 end
            elseif code == 28 then
                local t = tools[sel]
                if t then runOne(t) end
            elseif ch == 113 or ch == 81 then
                term.clear()
                term.setCursor(1, 1)
                return
            elseif ch and ch >= 49 and ch <= 57 then
                local want = tonumber(string.char(ch))
                if want and want <= #tools then sel = want end
            end
        end
    end
end
local function cliCtx()
    return {
        term = term,
        event = event,
        component = component,
        width = 1000,
        height = 1000,
        clear = function() end,
        write = function(s) print(tostring(s)) end,
        writeRaw = function(s) io.write(tostring(s)) end,
        ask = function(prompt, default)
            if prompt then io.write(tostring(prompt)) end
            local line = io.read()
            if line == nil or line == "" then return default end
            return line
        end,
        pause = function(msg)
            if msg then print(tostring(msg)) end
        end,
        title = function(s) print(tostring(s)) end,
    }
end
local function showAbout(tools, bad)
    local nEmb, nEmbTools = 0, 0
    for path in pairs(EMBEDDED) do
        nEmb = nEmb + 1
        local name = tostring(path):match("([^/]+)$") or ""
        if name:sub(1, 1) ~= "_" then nEmbTools = nEmbTools + 1 end
    end
    print("工具箱 " .. VERSION)
    print("")
    print(string.format("主程序内嵌工具 : %d 个", nEmb))
    print("工具目录       : " .. TOOL_DIR)
    print(string.format("实际加载工具   : %d 个", #tools))
    for _, t in ipairs(tools) do
        print(string.format("   %-18s %s", t.id, t.name))
    end
    if #bad > 0 then
        print("")
        print("加载失败：")
        for _, b in ipairs(bad) do print("   " .. b.file .. " —— " .. b.err) end
    end
    print("")
    print("版本指纹       : " .. embedFingerprint())
    print("")
    if nEmb == 0 then
        print("!! 这个主程序里没有内嵌工具——你跑的很可能是旧版文件")
        print("   重新下载最新的 toolbox 覆盖 /home/bin/toolbox")
    elseif nEmbTools ~= #tools then
        print("!! 内嵌 " .. nEmbTools .. " 个工具，但只加载到 " .. #tools .. " 个，可能有工具写失败了")
    else
        print("看起来正常。")
    end
end
local function runDiag()
    term.setCursorBlink(true)
    term.clear()
    term.setCursor(1, 1)
    print("事件诊断")
    print("----------------------------------------")
    print("现在按方向键、回车，或点屏幕，下面会实时显示收到的事件。")
    print("按 Q 退出。")
    print("如果这里什么都没出现，说明按键事件根本没到这台电脑：")
    print("  检查键盘是不是贴着屏幕、键盘有没有插好。")
    print("----------------------------------------")
    local n = 0
    while true do
        local e = { event.pull() }
        local name = tostring(e[1])
        n = n + 1
        if name == "key_down" then
            local ch, code = e[3], e[4]
            local chs = "无"
            if ch and ch >= 32 and ch < 127 then chs = string.char(ch) end
            print(string.format("#%-3d key_down   字符=%s  键码=%s", n, chs, tostring(code)))
            if ch == 113 or ch == 81 then break end
        elseif name == "touch" then
            print(string.format("#%-3d touch      列=%s  行=%s", n, tostring(e[3]), tostring(e[4])))
        elseif name ~= "timer" then
            print(string.format("#%-3d %s", n, name))
        else
            n = n - 1
        end
        if n > 300 then
            print("事件太多，先停下。")
            break
        end
    end
    term.setCursorBlink(true)
    print("")
    print("诊断结束。")
end
local function printList(tools, bad)
    print("工具箱 v" .. VERSION .. " —— 工具目录 " .. TOOL_DIR)
    print("")
    if #tools == 0 then
        print("（没有工具）")
    else
        local lastGroup = nil
        for _, t in ipairs(tools) do
            if t.group ~= lastGroup then
                lastGroup = t.group
                print("[" .. t.group .. "]")
            end
            print(string.format("  %-16s %s", t.id, t.name))
            if t.desc then
                print("      " .. t.desc)
            end
        end
    end
    if #bad and #bad > 0 then
        print("")
        print("以下工具加载失败：")
        for _, b in ipairs(bad) do
            print("  " .. b.file .. " —— " .. b.err)
        end
    end
end

end

    for _, t in ipairs(tools) do
        if t.id == cmd then target = t break end
    end
    if not target then
        print("没有这个工具：" .. cmd)
        print("")
        printList(tools, bad)
        return
    end
    local args = {}
    for i2 = 2, #argv do args[#args + 1] = argv[i2] end
    local ctx = cliCtx()
    local ok, err = pcall(target.run, ctx, args)
    if not ok then
        print("")
        print("!! 工具出错：" .. tostring(err))
    end
    return
end
