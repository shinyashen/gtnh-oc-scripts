local function fit(s, width)
    s = tostring(s or "")
    if width <= 0 then return "" end
    local ok, w = pcall(function() return unicode.wlen(s) end)
    if ok and w and w > width then
        local ok2, r = pcall(function() return unicode.wtrunc(s, width) end)
        if ok2 and r then return r end
        return s:sub(1, width)
    elseif not ok and #s > width then
        return s:sub(1, width)
    end
    return s
end
local function ulen(s)
    s = tostring(s or "")
    local ok, n = pcall(function() return unicode.wlen(s) end)
    return (ok and n) or #s
end
local function usub(str, a, b)
    str = tostring(str or "")
    local ok, r = pcall(function() return unicode.sub(str, a, b) end)
    return (ok and r) or str:sub(a, b)
end
local function wrapText(text, width, maxLines)
    text = tostring(text or "")
    local out, cur = {}, ""
    if width <= 0 then return out end
    local n = ulen(text)
    for i2 = 1, n do
        local ch = usub(text, i2, i2)
        if ch == "\n" then
            out[#out + 1] = cur
            cur = ""
        else
            local try = cur .. ch
            if ulen(try) > width then
                out[#out + 1] = cur
                cur = ch
            else
                cur = try
            end
        end
        if maxLines and #out >= maxLines then return out end
    end
    if cur ~= "" and (not maxLines or #out < maxLines) then out[#out + 1] = cur end
    return out
end
local function waitInput()
    while true do
        local e = { event.pull() }     -- 不带过滤器、不带超时：纯阻塞，最稳
        local name = e[1]
        if name == "key_down" then
            return "key_down", e[3], e[4]
        elseif name == "touch" then
            return "touch", e[3], e[4]
        end
    end
end
local function padRight(s, width)
    s = tostring(s or "")
    local ok, w = pcall(function() return unicode.wlen(s) end)
    w = (ok and w) or #s
    local pad = width - w
    if pad < 1 then pad = 1 end
    return s .. string.rep(" ", pad)
end

                hasTag = it.hasTag == true,
                oreDict = tostring(it.oreDict or ""),
            }
        end
    end
    return out
end
local function materialFromLabel(lb)
    lb = tostring(lb or "")
    if not lb:find("[Ii]ngot$") then return nil end
    local mat = lb:gsub("^Hot%s+", "")
    mat = mat:gsub("%s*[Ii]ngot$", "")
    mat = mat:lower()
    mat = mat:gsub("%s+", "")
    if mat == "" then return nil end
    return mat
end
local function ingotMaterial(it, labels)
    local lb0 = tostring(it.label or "")
    if lb0 ~= "" then
        local m0 = materialFromLabel(lb0)
        if m0 then return m0 end
    end
    if labels then
        local key = string.format("%s:%d", tostring(it.name or ""), tonumber(it.damage) or 0)
        local lb = labels[key]
        if lb and lb ~= "" then
            local m1 = materialFromLabel(lb)
            if m1 then return m1 end
        end
    end
    local od = tostring(it.oreDict or "")
    if od ~= "" then
        local m2 = od:match("^ingotHot(%a[%w]*)$") or od:match("^ingot(%a[%w]*)$")
        if m2 then return m2:lower() end
    end
    local n = tostring(it.name or ""):lower()
    local m3 = n:match("([%w_]+)_hot_ingot$") or n:match("([%w_]+)_ingot$")
    if m3 then return m3 end
    return nil
end
local function isIngot(e)
    if e.fluid then return false end
    local lb = tostring(e.label or "")
    if lb:find("[Ii]ngot$") then return true end
    if lb == "" then
        local od = tostring(e.oreDict or "")
        if od:sub(1, 5) == "ingot" then return true end
        local n = tostring(e.name or ""):lower()
        if n:match("_ingot$") or n:match(":ingot") then return true end
    end
    return false
end
local function fluidMaterial(fname)
    local n = tostring(fname or ""):lower()
    local m = n:match("molten[%._]([%w_]+)$")
    if m then return m end
    m = n:match("^fluid%.([%w_]+)$")
    if m then return m end
    return nil
end
local function learnLabels(iface)
    local map = {}
    local f = iface.getItemsInNetwork
    if not f then return map, false end
    local ok, list = pcall(f)
    if not ok or type(list) ~= "table" then return map, false end
    local n = 0
    for _, it in ipairs(list) do
        if type(it) == "table" and it.name then
            local k = string.format("%s:%d", tostring(it.name), tonumber(it.damage) or 0)
            if it.label and tostring(it.label) ~= "" and map[k] == nil then
                map[k] = tostring(it.label)
                n = n + 1
            end
        end
    end
    return map, n > 0
end
local function learnFluidNames(iface)
    local map = {}
    for slot = 1, 36 do
        local ok, p = pcall(iface.getInterfacePattern, slot)
        if ok and p then
            for _, list in ipairs({ p.outputs, p.inputs }) do
                for _, k in ipairs(sortedKeys(list)) do
                    local it = list[k]
                    if type(it) == "table" and isFluid(it) then
                        local mat = fluidMaterial(it.name)
                        if mat and map[mat] == nil then
                            map[mat] = tostring(it.name)
                        end
                    end
                end
            end
        end
    end
    return map
end
local function keyOfEntry(e)
    return table.concat({ tostring(e.name or ""), tostring(e.damage or 0), tostring(e.size or 1) }, "|")
end
local function convertOutputs(outs, fluidMap, manual, known, labels)
    local newOuts, log, skipped = {}, {}, {}
    for _, e in ipairs(outs) do
        if e.fluid then
            newOuts[#newOuts + 1] = e
            log[#log + 1] = { kind = "keepFluid", from = e, to = e }
        elseif isIngot(e) then
            local mat = ingotMaterial(e, labels)
            local fname = (manual and manual[entryKey(e)])
                or (known and known[entryKey(e)])
                or (mat and fluidMap[mat]) or nil
            if not fname and mat and not fluidMap[mat] then
                fname = FLUID_PREFIX .. mat
            end
            if fname then
                local n = (tonumber(e.size) or 1) * MB_PER_INGOT
                local ne = {
                    name = fname,
                    damage = 0,
                    size = n,
                    fluid = true,
                    label = "",
                    hasTag = false,
                }
                newOuts[#newOuts + 1] = ne
                log[#log + 1] = { kind = "converted", from = e, to = ne, mat = mat }
            else
                newOuts[#newOuts + 1] = e
                skipped[#skipped + 1] = e
                log[#log + 1] = { kind = "fail", from = e, to = e }
            end
        else
            newOuts[#newOuts + 1] = e
            log[#log + 1] = { kind = "keepOther", from = e, to = e }
        end
    end
    return newOuts, log, skipped
end
local function dumpEntry(it)
    if type(it) ~= "table" then return tostring(it) end
    local ks = {}
    for k in pairs(it) do ks[#ks + 1] = tostring(k) end
    table.sort(ks)
    local parts = {}
    for _, k in ipairs(ks) do
        local v = it[k]
        local vs
        if type(v) == "table" then vs = "<table>" else vs = tostring(v) end
        parts[#parts + 1] = k .. "=" .. vs
    end
    return table.concat(parts, " ")
end
local function writeOne(iface, kind, slot, idx, e)
    local candidates
    if e.fluid then
        local short = e.name
        local long = e.name
        if short:sub(1, 9) == "gregtech:" then
            short = short:sub(10)
        else
            long = "gregtech:" .. short
        end
        candidates = {
            { data = { name = short, size = e.size }, tag = "fluid", label = "size/短名" },
            { data = { name = long, size = e.size }, tag = "fluid", label = "size/长名" },
            { data = { name = short, amount = e.size }, tag = "fluid", label = "amount/短名" },
            { data = { name = long, amount = e.size }, tag = "fluid", label = "amount/长名" },
        }
    else
        local d = { name = e.name, size = e.size }
        if e.damage and e.damage ~= 0 then d.damage = e.damage end
        if e.tag then d.tag = e.tag end
        candidates = { { data = d, tag = "item", label = "item" } }
    end
    local lastErr = nil
    for _, c in ipairs(candidates) do
        pcall(function()
            if kind == "out" then
                return iface.setInterfacePatternOutput(slot, idx, c.data, c.tag)
            else
                return iface.setInterfacePatternInput(slot, idx, c.data, c.tag)
            end
        end)
        os.sleep(0.15)
        local okR, pat = pcall(iface.getInterfacePattern, slot)
        if okR and pat then
            local list = (kind == "out") and pat.outputs or pat.inputs
            local got = list and list[idx]
            local gotName = got and tostring(got.name) or nil
            if gotName == e.name then
                return true, c.label
            end
            if gotName == tostring(c.data.name) then
                return true, c.label
            end
            if gotName == nil then
                return true, c.label .. "(未回读)"
            end
            lastErr = string.format("%s: 回读是 %s", c.label, gotName)
        else
            return true, c.label .. "(未回读)"
        end
    end
    return false, lastErr or "四种写法都试过了"
end
local function writePattern(iface, slot, outs, ins, dry)
    local notes = {}
    local function put(kind, list)
        for idx, e in ipairs(list) do
            if dry then
                notes[#notes + 1] = string.format("%s#%d 跳过(预览)", kind, idx)
            else
                local ok, how = writeOne(iface, kind, slot, idx, e)
                if not ok then
                    return false, string.format("%s 第 %d 项失败：%s", kind, idx, tostring(how))
                end
                notes[#notes + 1] = string.format("%s#%d=%s", kind, idx, tostring(how))
            end
            os.sleep(0)
        end
        return true
    end
    local ok1, err1 = put("in", ins)
    if not ok1 then return false, err1 end
    local ok2, err2 = put("out", outs)
    if not ok2 then return false, err2 end
    if not dry then
        local chk = iface.getInterfacePattern(slot)
        if chk then
            for kind, want in pairs({ ["in"] = #ins, ["out"] = #outs }) do
                local list = (kind == "in") and chk.inputs or chk.outputs
                local n = 0
                if list then for _ in pairs(list) do n = n + 1 end end
                if n > want then
                    notes[#notes + 1] = string.format("警告：%s 还有 %d 项多余的老条目（%d 应为 %d）",
                        kind, n - want, n, want)
                end
            end
        end
    end
    return true, table.concat(notes, " "), true
end
local function fmtEntry(e)
    return string.format("%s ×%s", brief(e, 10), numStr(e.size))
end

            end

            footer = "失败的会写回原样，可以再跑一次",
        })

    end

local function fluidCandidates(mat)
    local out = {}
    local key = mat:lower():gsub("[^%w]", "")
    if ALIASES[key] then
        out[#out + 1] = ALIASES[key]
    end
    local a = mat:lower():gsub("%s+", "_")
    local b = mat:lower():gsub("%s+", "")
    out[#out + 1] = "molten." .. a
    if b ~= a then out[#out + 1] = "molten." .. b end
    return out
end
local function findIface()
    for addr in component.list("fluid_interface") do
        return component.proxy(addr), "fluid_interface"
    end
    for addr in component.list("me_interface") do
        return component.proxy(addr), "me_interface"
    end
    return nil, nil
end
function M.run(ctx, args)
    local iface = findIface()
    if not iface then
        if ctx.select then
            ctx.select({ { label = "知道了", value = 1 } }, {
                title = "找不到ME接口", footer = "检查适配器是否紧贴方块",
            })
        else
            ctx.write("找不到ME接口")
        end
        return false
    end
    ctx.clear()
    ctx.write("压模改固化：扫描全部样板")
    ctx.write(string.rep("-", 56))
    local nDone, nSkip, nFail = 0, 0, 0
    for slot = 1, 36 do
        local okP, p = pcall(iface.getInterfacePattern, slot)
        if okP and p and p.outputs then
            local best = nil
            for _, e in pairs(p.outputs) do
                local lb = tostring(e.label or "")
                if lb ~= "" and not tostring(e.name or ""):lower():find("molten") then
                    local s = matchShape(lb)
                    if s then
                        best = { e = e, shape = s, label = lb }
                        break
                    end
                end
            end
            if best then
                local mat = materialOf(best.label)
                local need = (tonumber(best.e.size) or 1) * best.shape.l
                local targetIdx, targetLabel, alreadyFluid = nil, nil, false
                if p.inputs then
                    for idx, e in pairs(p.inputs) do
                        local inLabel = tostring(e.label or "")
                        local inName = tostring(e.name or ""):lower()
                        if inLabel:lower():find(mat:lower(), 1, true) then
                            if inName:find("molten") or inName:find("fluid") then
                                alreadyFluid = true
                                targetLabel = inLabel
                            else
                                targetIdx, targetLabel = idx, inLabel
                            end
                            break
                        end
                    end
                end
                if alreadyFluid then
                    nSkip = nSkip + 1
                elseif not targetIdx then
                    nSkip = nSkip + 1
                    ctx.write(string.format("槽%d %s：输入里没有含「%s」的项",
                        slot, best.shape.cn, mat))
                else
                    local done = false
                    for _, fname in ipairs(fluidCandidates(mat)) do
                        local okc = pcall(function()
                            return iface.setInterfacePatternInput(slot, targetIdx,
                                { name = fname, size = need }, "fluid")
                        end)
                        if okc then
                            done = true
                            break
                        end
                    end
                    if done then
                        nDone = nDone + 1
                        ctx.write(string.format("槽%d %s ×%s → %s ×%d",
                            slot, best.shape.cn, tostring(best.e.size),
                            fluidCandidates(mat)[1], need))
                    else
                        nFail = nFail + 1
                        ctx.write(string.format("槽%d %s：流体名 %s 不存在",
                            slot, best.shape.cn, fluidCandidates(mat)[1]))
                    end
                end
            end
        end
        os.sleep(0)
    end
    ctx.write(string.rep("-", 56))
    ctx.write(string.format("完成：改了 %d 个，跳过 %d 个，失败 %d 个", nDone, nSkip, nFail))
    if ctx.select then
        ctx.select({ { label = "好了", value = 1 } }, {
            title = string.format("压模改固化：改 %d / 跳过 %d / 失败 %d", nDone, nSkip, nFail),
            footer = "输入已经是流体的样班会自动跳过",
        })
    end
    return nFail == 0
end

local function findIfaces()
    local list = {}
    for addr in component.list("fluid_interface") do
        list[#list + 1] = component.proxy(addr)
    end
    if #list == 0 then
        for addr in component.list("me_interface") do
            list[#list + 1] = component.proxy(addr)
        end
    end
    return list
end
local function findDb()
    for addr in component.list("database") do
        return component.proxy(addr)
    end
    return nil
end
local function usedSlots(ifaces)
    local out = {}
    for _, iface in ipairs(ifaces) do
        for slot = 1, 36 do
            local ok, p = pcall(iface.getInterfacePattern, slot)
            if ok and p then
                out[#out + 1] = { iface = iface, slot = slot }
            end
        end
    end
    return out
end

end
local function writeOne(s, w, db)
    local nErr = nil
    local iface, slot = s.iface, s.slot
    for i, o in ipairs(w.outs) do
        local ok, err
        if o.rename and db then
            ok, err = pcall(db.set, 1, o.name, o.damage or 0, '{display:{Name:"' .. o.rename .. '"}}')
            if ok then
                ok, err = pcall(iface.setInterfacePatternOutput, slot, i, db.address, 1, o.size or 1)
            end
        elseif o.fluid then
            ok, err = pcall(iface.setInterfacePatternOutput, slot, i, { name = o.name, size = o.size }, "fluid")
        else
            ok, err = pcall(iface.setInterfacePatternOutput, slot, i,
                { name = o.name, size = o.size, damage = o.damage or 0 }, "item")
        end
        if not ok then nErr = tostring(err) end
    end
    for i, e in ipairs(w.ins) do
        local ok, err
        if e.rename and e.renameTo and db then
            ok, err = pcall(db.set, 1, e.name, e.damage or 0, '{display:{Name:"' .. e.renameTo .. '"}}')
            if ok then
                ok, err = pcall(iface.setInterfacePatternInput, slot, i, db.address, 1, e.size)
            end
        elseif e.fluid then
            ok, err = pcall(iface.setInterfacePatternInput, slot, i, { name = e.name, size = e.size }, "fluid")
        else
            ok, err = pcall(iface.setInterfacePatternInput, slot, i,
                { name = e.name, size = e.size, damage = e.damage or 0 }, "item")
        end
        if not ok then nErr = tostring(err) end
    end
    return nErr
end

local function genNAC(idx, renameTo)
    local r = RECIPES[idx]
    local ins = {}
    for _, e in ipairs(r.ins) do
        local t = item(e.name, e.size, e.damage)
        t.fluid = e.fluid
        if e.rename then
            t.rename = true
            t.renameTo = renameTo or e.label
        end
        ins[#ins + 1] = t
    end
    local outs = {}
    for _, o in ipairs(r.out) do
        outs[#outs + 1] = item(o.name, o.size, o.damage)
    end
    return { { tag = r.name, outs = outs, ins = ins } }
end

local function runList(ctx, ifaces, list, db)
    local total = #list
    local idx = 1
    local nOk, nErr = 0, 0
    ctx.clear()
    ctx.write(string.format("共 %d 个样板，分批写入", total))
    ctx.write("每批写满当前所有占位样板后暂停，换好再继续")
    ctx.write("")
    while idx <= total do
        local slots = usedSlots(ifaces)
        if #slots == 0 then
            if not ctx.select then
                ctx.write("没有可写的样板槽")
                return false
            end
            local go = ctx.select({
                { label = "已放好，继续", value = true },
                { label = "退出", value = false },
            }, {
                title = string.format("接口里没有占位样板（还剩 %d 个没写）", total - idx + 1),
                footer = "把占位样板放进接口，再选「已放好，继续」",
            })
            if not go then return true end
        else
            local batch = math.min(#slots, total - idx + 1)
            for j = 1, batch do
                local w = list[idx + j - 1]
                local err = writeOne(slots[j], w, db)
                if err then
                    nErr = nErr + 1
                    ctx.write(string.format("  [%d] %s 失败: %s", idx + j - 1, w.tag, err))
                else
                    nOk = nOk + 1
                end
                os.sleep(0)
            end
            idx = idx + batch
            if idx <= total then
                local go = true
                if ctx.select then
                    go = ctx.select({
                        { label = "已换好，继续", value = true },
                        { label = "退出", value = false },
                    }, {
                        title = string.format("已写 %d/%d，还剩 %d 个", idx - 1, total, total - idx + 1),
                        footer = "把这批样板拿走、放新的占位样板，再选「已换好，继续」",
                    })
                elseif ctx.pause then
                    ctx.pause("把这批样板拿走、放新的占位样板，按任意键继续")
                end
                if not go then return true end
            end
        end
    end
    ctx.write("")
    ctx.write(string.format("完成：成功 %d，失败 %d", nOk, nErr))
    if ctx.pause then ctx.pause("") end
    return nErr == 0
end

function M.run(ctx, args)
    local ifaces = findIfaces()
    if #ifaces == 0 then
        say(ctx, "找不到ME接口", "检查适配器是否紧贴 ME 接口")
        return false
    end
    local slots = usedSlots(ifaces)
    if #slots == 0 then
        say(ctx, "接口里没有样板", "先往接口里塞占位样板，工具只覆盖已有样板的槽")
        return false
    end
    local pick = menu(ctx, {
        { label = "皮米电路", desc = "NAC · 31 项输入", value = "nac:1" },
        { label = "量子电路", desc = "NAC · 36 项输入", value = "nac:2" },
        { label = "神锻等离子样板", desc = "94 种 → 等离子", value = "plasma" },
        { label = "夸克 · 发配样板", desc = "粉+1296 / 气体+1000 → 单元", value = "quark:fapei" },
        { label = "夸克 · 汇总下单", desc = "80 单元 ×2e9 → 铁剑", value = "quark:sum" },
        { label = "夸克 · 等离子补货", desc = "80 等离子×权重 → 生鳕鱼", value = "quark:refill" },
        { label = "磁物质 · 发配样板", desc = "14 种材料 + 富快子捆绑", value = "ci:fapei" },
        { label = "磁物质 · 纯粉发配", desc = "14 种粉 → 对应锭", value = "ci:powder" },
        { label = "磁物质 · 汇总下单", desc = "2 个：锭→金剑 / 单元→钻石剑", value = "ci:sum" },
        { label = "磁物质 · 等离子补货", desc = "14 等离子 → 生鳕鱼", value = "ci:refill" },
    }, string.format("样板生成：%d 个接口 / %d 个可用槽", #ifaces, #slots))
    if not pick then return true end
    local group, key = pick:match("^([^:]+):(.+)$")
    local list
    if group == "nac" then
        local idx = tonumber(key)
        local renameTo = nil
        local hasRename = false
        for _, e in ipairs(RECIPES[idx].ins) do
            if e.rename then hasRename = true end
        end
        if hasRename then
            ctx.clear()
            ctx.write("重命名物品")
            ctx.write("")
            ctx.write("这个样板里有几个物品需要统一改名，输入名字就行。")
            ctx.write("回车确定，留空则用原名")
            ctx.write("")
            renameTo = ctx.ask("名字> ", "")
            if renameTo == "" then renameTo = nil end
            ctx.clear()
        end
        list = genNAC(idx, renameTo)
    elseif pick == "plasma" then
        list = genPlasma()
    elseif group == "quark" then
        list = genQuark(key)
    elseif group == "ci" then
        list = genCI(key)
    end
    if not list or #list == 0 then return true end
    local db = findDb()
    if (group == "quark" or group == "ci") and not db then
        local needRename = false
        for _, w in ipairs(list) do
            for _, o in ipairs(w.outs) do
                if o.rename then needRename = true end
            end
            for _, e in ipairs(w.ins) do
                if e.rename then needRename = true end
            end
        end
        if needRename then
            say(ctx, "需在适配器中放入数据库升级", "要写重命名物品，需在适配器中放入数据库升级")
            return false
        end
    end
    return runList(ctx, ifaces, list, db)
end

local STAMP = TOOL_DIR .. "/.installed"
local function embedFingerprint()
    local n = 0
    for path, body in pairs(EMBEDDED) do
        n = n + #path + #body
    end
    return tostring(n)
end
local function bootstrapTools()
    local fp = embedFingerprint()
    local old = ""
    local okR = pcall(function()
        local fh = io.open(STAMP, "r")
        if not fh then return end
        old = fh:read("*a") or ""
        fh:close()
    end)
    if not okR then old = "" end
    old = old:gsub("%s+$", "")
    local missing = 0
    for path in pairs(EMBEDDED) do
        if not fs.exists(path) then missing = missing + 1 end
    end
    if missing == 0 and old == fp then
        return false, 0
    end
    if not fs.isDirectory(TOOL_DIR) then
        pcall(function() fs.makeDirectory(TOOL_DIR) end)
    end
    local n = 0
    for path, body in pairs(EMBEDDED) do
        local ok, f = pcall(io.open, path, "w")
        if ok and f then
            f:write(body)
            f:close()
            local chk = io.open(path, "r")
            local back = chk and chk:read("*a") or ""
            if chk then chk:close() end
            if back == body then n = n + 1 end
        end
    end
    local ok2, sf = pcall(io.open, STAMP, "w")
    if ok2 and sf then
        sf:write(fp)
        sf:close()
    end
    return n > 0, n
end
local function loadTools()
    local list, bad = {}, {}
    if not fs.isDirectory(TOOL_DIR) then
        return list, bad, "目录不存在：" .. TOOL_DIR
    end
    for file in fs.list(TOOL_DIR) do
        local name = tostring(file)
        if name ~= "." and name ~= ".." and name:sub(1, 1) ~= "_" and name:sub(-4) == ".lua" then
            local path = TOOL_DIR .. "/" .. name
            local ok, mod = pcall(function()
                local chunk = loadfile(path)
                if not chunk then return nil end
                return chunk()
            end)
            if ok and type(mod) == "table" and type(mod.run) == "function" then
                mod._file = name
                mod.id = mod.id or name:sub(1, -5)
                mod.name = mod.name or mod.id
                mod.group = mod.group or "其他"
                mod.order = tonumber(mod.order) or 100
                list[#list + 1] = mod
            else
                bad[#bad + 1] = { file = name, err = tostring(mod) }
            end
        end
    end
    table.sort(list, function(a, b)
        if a.group ~= b.group then return a.group < b.group end
        if a.order ~= b.order then return a.order < b.order end
        return a.name < b.name
    end)
    return list, bad
end
local HITS = {}
local function put(x, y, text, fgRole, bgRole)
    if y < 1 or y > H or x > W then return end
    term.setCursor(x, y)
    useFg(fgRole or "text")
    useBg(bgRole or "canvas")
    term.write(fit(text, W - x + 1))
end
local function hook(x, y, text, action)
    HITS[#HITS + 1] = { x1 = x, x2 = x + ulen(text) - 1, y = y, action = action }
end
local function hitAt(x, y)
    for _, h in ipairs(HITS) do
        local y1 = h.y1 or h.y
        local y2 = h.y2 or h.y
        if y >= y1 and y <= y2 and x >= h.x1 and x <= h.x2 then
            return h.action
        end
    end
    return nil
end
local function drawCard(x, y, w, h, t, isSel)
    local bg = isSel and "selected" or "surface"
    local bd = isSel and "borderOn" or "border"
    useBg(bg)
    useFg("text")
    for r = 0, h - 1 do
        term.setCursor(x, y + r)
        term.write(string.rep(" ", w))
    end
    useFg(bd)
    useBg(bg)
    term.setCursor(x, y)
    term.write("┌" .. string.rep("─", w - 2) .. "┐")
    for r = 1, h - 2 do
        term.setCursor(x, y + r); term.write("│")
        term.setCursor(x + w - 1, y + r); term.write("│")
    end
    term.setCursor(x, y + h - 1)
    term.write("└" .. string.rep("─", w - 2) .. "┘")
    local inner = w - 4
    put(x + 2, y + 1, t.name, isSel and "accent" or "text", bg)
    local descLines = wrapText(t.desc or "", inner, h - 3)
    for k, ln in ipairs(descLines) do
        put(x + 2, y + 1 + k, ln, "muted", bg)
    end
    return inner
end
local function makeCtx(opts)
    opts = opts or {}
    local ctx
    local function resetScreen()
        term.setCursorBlink(false)
        term.clear()
        term.setCursor(1, 1)
        lastFg, lastBg = nil, nil
        useBg("canvas")
        HITS = {}
    end
    local function doSelect(items, o)
        o = o or {}
        local sel = tonumber(o.default) or 1
        local head = o.title or "请选择"
        local foot = o.footer or "点一下选中   ↑↓ 移动   Enter 确定   Q 返回（别按 Esc，那是游戏菜单）"
        local top = o.top or 3
        local bottom = H - 2
        while true do
            if ctx then ctx.hadSelect = true end
            resetScreen()
            put(1, 1, "  " .. head, "accent", "surface")
            put(1, 2, string.rep("-", W), "border", "canvas")
            local y = top
            for i2, it in ipairs(items) do
                if y + 1 > bottom then break end
                local isSel = (i2 == sel)
                local bg = isSel and "selected" or "canvas"
                term.setCursor(1, y)
                useBg(bg)
                pcall(function() term.clearLine() end)
                local mark = isSel and "> " or "  "
                put(1, y, mark .. tostring(it.label or ""), isSel and "accent" or "text", bg)
                if it.desc and tostring(it.desc) ~= "" then
                    local d = tostring(it.desc)
                    local dx = math.floor(W / 2)
                    if dx > ulen(mark .. tostring(it.label or "")) + 2 then
                        put(dx, y, d, "muted", bg)
                    end
                end
                term.setCursor(1, y + 1)
                useBg(bg)
                pcall(function() term.clearLine() end)
                HITS[#HITS + 1] = {
                    x1 = 1, x2 = W, y1 = y, y2 = y + 1,
                    action = { kind = "pick", idx = i2 },
                }
                y = y + 2
            end
            put(1, H - 1, string.rep("-", W), "border", "canvas")
            local btxt = "[ 返回 ]"
            put(1, H, btxt, "muted", "surface")
            HITS[#HITS + 1] = { x1 = 1, x2 = ulen(btxt), y1 = H, y2 = H,
                                action = { kind = "back" } }
            put(ulen(btxt) + 2, H, foot, "muted", "canvas")
            term.setCursor(1, H)
            local ev, a, b = waitInput()
            if ev == "touch" then
                local act = hitAt(a, b)
                if act and act.kind == "pick" then
                    local it = items[act.idx]
                    if it then return it.value, it end
                elseif act and act.kind == "back" then
                    return nil
                end
            elseif ev == "key_down" then
                local ch, code = a, b
                if code == 200 then
                    sel = math.max(1, sel - 1)
                elseif code == 208 then
                    sel = math.min(#items, sel + 1)
                elseif code == 28 then
                    local it = items[sel]
                    if it then return it.value, it end
                elseif ch == 113 or ch == 81 then
                    return nil
                elseif ch and ch >= 49 and ch <= 57 then
                    local n2 = tonumber(string.char(ch))
                    if n2 and items[n2] then return items[n2].value, items[n2] end
                end
            end
        end
    end
    local function doPrompt(question, default, hint)
    local buf = ""
    while true do
        resetScreen()
        HITS = {}
        put(1, 1, "  " .. tostring(question), "accent", "surface")
        put(1, 2, string.rep("-", W), "border", "canvas")
        local shown = (buf ~= "") and buf or "（还没输入，直接回车就用默认）"
        put(1, 4, "  当前：" .. shown, "text", "canvas")
        put(1, 5, "  回车 = 用 " .. (default ~= nil and ("「" .. tostring(default) .. "」") or "空"),
            "muted", "canvas")
        put(1, H - 1, string.rep("-", W), "border", "canvas")
        local bx = 1
        local function btn(label, act, fg)
            local txt = "[ " .. label .. " ]"
            put(bx, H, txt, fg or "accent", "surface")
            HITS[#HITS + 1] = { x1 = bx, x2 = bx + ulen(txt) - 1, y1 = H, y2 = H, action = act }
            bx = bx + ulen(txt) + 1
        end
        btn("确定", { kind = "ok" })
        btn("取消", { kind = "cancel" }, "muted")
        put(bx + 1, H, "  打字输入，Enter 或点「确定」提交", "muted", "canvas")
        term.setCursorBlink(true)
        term.setCursor(1 + ulen("  当前：") + ulen(shown) + 1, 4)
        local ev, a, b = waitInput()
        if ev == "touch" then
            local act = hitAt(a, b)
            if act then
                term.setCursorBlink(false)
                if act.kind == "ok" then
                    if buf ~= "" then return buf end
                    return default
                elseif act.kind == "cancel" then
                    return nil
                end
            end
        elseif ev == "key_down" then
            local ch, code = a, b
            if code == 28 then
                term.setCursorBlink(false)
                if buf ~= "" then return buf end
                return default
            elseif code == 14 then
                if #buf > 0 then buf = buf:sub(1, #buf - 1) end
            elseif ch and ch >= 32 and ch < 127 then
                if #buf < 40 then buf = buf .. string.char(ch) end
            end
        end
    end
end

            { label = yesHint or "确定", value = true },
            { label = noHint or "取消", value = false },

            title = question,
            default = 1,
            footer = "点按钮选择   ↑↓ 移动   Enter 确定   Q = 取消",
        })
        return v == true
    end
    local function doCardSelect(items, o)

                    HITS[#HITS + 1] = {
                        x1 = 3, x2 = 3 + cw - 1, y1 = y, y2 = y + ch - 1,
                        action = { kind = "pick", idx = i },
                    }
                end
            end
            put(1, H - 1, string.rep("─", W), "border", "canvas")
            local btxt = "[ 返回 ]"
            put(1, H, btxt, "muted", "surface")
            HITS[#HITS + 1] = { x1 = 1, x2 = ulen(btxt), y1 = H, y2 = H, action = { kind = "back" } }
            put(ulen(btxt) + 2, H, "↑↓ 选择   Enter 确定   Q 返回", "muted", "canvas")
            local ev, a, b = waitInput()
            if ev == "touch" then
                local act = hitAt(a, b)
                if act and act.kind == "pick" then
                    local it = items[act.idx]
                    if it then return it.value or it.label, it end
                elseif act and act.kind == "back" then
                    return nil
                end
            elseif ev == "key_down" then
                local ch2, code = a, b
                if code == 200 then
                    sel = math.max(1, sel - 1)
                elseif code == 208 then
                    sel = math.min(#items, sel + 1)
                elseif code == 28 then
                    local it = items[sel]
                    if it then return it.value or it.label, it end
                elseif ch2 == 113 or ch2 == 81 then
                    return nil
                end
            end
        end
    end
    ctx = {
        term = term,
        event = event,
        component = component,
        width = W,
        height = H,
        clear = function()
            resetScreen()
        end,
        write = function(s2)
            useFg("text")
            term.write(fit(s2, W))
            term.write("\n")
        end,
        writeRaw = function(s2)
            term.write(fit(s2, W))
        end,
        ask = function(prompt, default)
            if prompt then term.write(fit(prompt, W)) end
            local ok, line2 = pcall(function() return term.read() end)
            if not ok or line2 == nil or line2 == "" then return default end
            return line2
        end,
        pause = function(msg)
            if msg then term.write(fit(msg, W)) end
            term.write(fit("按任意键继续……", W))
            waitInput()
        end,
        title = function(s2)
            local x, y = term.getCursor()
            put(1, 1, "  " .. tostring(s2), "accent", "surface")
            term.setCursor(x, y)
        end,
        select = doSelect,
        prompt = doPrompt,
        confirm = doConfirm,
        cardSelect = doCardSelect,
        hr = function(y)
            put(1, y, string.rep("-", W), "border", "canvas")
        end,
        line = function(x, y, s2, fgRole, bgRole)
            put(x or 1, y, s2, fgRole, bgRole)
        end,
    }
    return ctx
end
local COLS = 2
local CARD_H = 5
local function drawMenu(tools, bad, sel, scroll, status)
    term.setCursorBlink(false)
    HITS = {}
    term.clear()
    lastFg, lastBg = nil, nil
    useBg("canvas")
    local total = tostring(#tools) .. " 个工具"
    put(1, 1, "  工具箱", "accent", "canvas")
    put(1 + ulen("  工具箱") + 1, 1, "v" .. VERSION, "muted", "canvas")
    put(W - ulen(total), 1, total, "muted", "canvas")
    put(1, 2, string.rep("─", W), "border", "canvas")
    local top = 4
    local bottom = H - 3
    local gapX = 2
    local cw = math.floor((W - gapX * (COLS + 1)) / COLS)
    if cw < 16 then cw = 16 end
    local stepY = CARD_H + 1
    local rowsVisible = math.max(1, math.floor((bottom - top + 1) / stepY))
    if #tools == 0 then
        put(2, top, "一个工具都没有", "warn", "canvas")
        put(2, top + 1, "把 .lua 工具文件放进 " .. TOOL_DIR .. "/", "muted", "canvas")
        put(2, top + 2, "或者删掉主程序重新下载（它会自动补齐）", "muted", "canvas")
    else
        for i, t in ipairs(tools) do
            local slot = i - 1 - scroll * COLS
            local row = math.floor(slot / COLS)
            local col = slot % COLS
            if row >= 0 and row < rowsVisible then
                local x = 1 + gapX + col * (cw + gapX)
                local y = top + row * stepY
                drawCard(x, y, cw, CARD_H, t, i == sel)
                HITS[#HITS + 1] = {
                    x1 = x, x2 = x + cw - 1,
                    y1 = y, y2 = y + CARD_H - 1,
                    action = { kind = "tool", idx = i },
                }
            end
        end
    end
    if #bad and #bad > 0 then
        put(1, H - 2, fit(string.format("  !! %d 个工具加载失败：%s", #bad, bad[1].file), W),
            "error", "canvas")
    elseif status and status ~= "" then
        put(1, H - 2, "  " .. status, "warn", "canvas")
    else
        put(1, H - 2, string.rep("─", W), "border", "canvas")
    end
    local bx = 1
    local function button(label, act, fgRole)
        local txt = "[ " .. label .. " ]"
        put(bx, H - 1, txt, fgRole or "accent", "surface")
        HITS[#HITS + 1] = { x1 = bx, x2 = bx + ulen(txt) - 1, y1 = H - 1, y2 = H - 1, action = act }
        bx = bx + ulen(txt) + 1
    end
    button("运行", { kind = "run" })
    button("重扫", { kind = "rescan" })
    button("退出", { kind = "quit" }, "muted")
    put(bx + 1, H - 1, "↑↓←→ 选择   回车运行   也可直接点卡片", "muted", "canvas")
    term.setCursor(1, H - 1)
end
local function menuHeight(tools)
    local h2, lastG = 0, nil
    for i3 = 1, #tools do
        if tools[i3].group ~= lastG then
            lastG = tools[i3].group
            h2 = h2 + 1
        end
        h2 = h2 + 1
    end
    return h2
end
local function runMenu(tools, bad)
    local sel, scroll = 1, 0
    local status = ""
    local function runOne(t)
        term.setCursorBlink(false)
        term.clear()
        term.setCursor(1, 1)
        local ctx = makeCtx()
        local ok, err = pcall(t.run, ctx, {})
        if not ok then
            term.setCursor(1, H)
            term.write(fit("!! 工具出错：" .. tostring(err), W))
            term.setCursor(1, H)
            term.write(fit("[按任意键返回菜单]", W))
            waitInput()
        elseif not ctx.hadSelect then
            term.setCursor(1, H)
            term.write(fit("[按任意键返回菜单]", W))
            waitInput()
        end
    end
    while true do
        local top, bottom = 4, H - 3
        local stepY = CARD_H + 1
        local rowsVisible = math.max(1, math.floor((bottom - top + 1) / stepY))
        local selRow = math.floor((sel - 1) / COLS)
        if selRow < scroll then scroll = selRow end
        if selRow >= scroll + rowsVisible then scroll = selRow - rowsVisible + 1 end
        if scroll < 0 then scroll = 0 end
        drawMenu(tools, bad, sel, scroll, status)
        status = ""
        local ev, a, b = waitInput()
        if ev == "touch" then
            local act = hitAt(a, b)
            if act then
                if act.kind == "tool" then
                    if act.idx == sel then
                        local t = tools[sel]
                        if t then runOne(t) end
                    else
                        sel = act.idx
                    end
                elseif act.kind == "run" then
                    local t = tools[sel]
                    if t then runOne(t) end
                elseif act.kind == "rescan" then
                    tools, bad = loadTools()
                    if sel > #tools then sel = math.max(1, #tools) end
                    status = string.format("已重扫：%d 个工具", #tools)
                elseif act.kind == "quit" then
                    term.clear()
                    term.setCursor(1, 1)
                    return
                end
            end
        elseif ev == "key_down" then
            local ch, code = a, b
            if code == 200 then
                if sel - COLS >= 1 then sel = sel - COLS end
            elseif code == 208 then
                if sel + COLS <= #tools then sel = sel + COLS end
            elseif code == 203 then
                if (sel - 1) % COLS > 0 then sel = sel - 1 end
            elseif code == 205 then
                if (sel - 1) % COLS < COLS - 1 and sel < #tools then sel = sel + 1 end
            elseif code == 28 then
                local t = tools[sel]
                if t then runOne(t) end
            elseif ch == 113 or ch == 81 then
                term.clear()
                term.setCursor(1, 1)
                return
            elseif ch and ch >= 49 and ch <= 57 then
                local want = tonumber(string.char(ch))
                if want and want <= #tools then sel = want end
            end
        end
    end
end

end

    end

        print("!! 内嵌 " .. nEmbTools .. " 个工具，但只加载到 " .. #tools .. " 个，可能有工具写失败了")
    else
        print("看起来正常。")
    end
end
local function runDiag()
    term.setCursorBlink(true)
    term.clear()
    term.setCursor(1, 1)
    print("事件诊断")
    print("----------------------------------------")
    print("现在按方向键、回车，或点屏幕，下面会实时显示收到的事件。")
    print("按 Q 退出。")
    print("如果这里什么都没出现，说明按键事件根本没到这台电脑：")
    print("  检查键盘是不是贴着屏幕、键盘有没有插好。")
    print("----------------------------------------")
    local n = 0
    while true do
        local e = { event.pull() }
        local name = tostring(e[1])
        n = n + 1
        if name == "key_down" then
            local ch, code = e[3], e[4]
            local chs = "无"
            if ch and ch >= 32 and ch < 127 then chs = string.char(ch) end
            print(string.format("#%-3d key_down   字符=%s  键码=%s", n, chs, tostring(code)))
            if ch == 113 or ch == 81 then break end
        elseif name == "touch" then
            print(string.format("#%-3d touch      列=%s  行=%s", n, tostring(e[3]), tostring(e[4])))
        elseif name ~= "timer" then
            print(string.format("#%-3d %s", n, name))
        else
            n = n - 1
        end
        if n > 300 then
            print("事件太多，先停下。")
            break
        end
    end
    term.setCursorBlink(true)
    print("")
    print("诊断结束。")
end
local function printList(tools, bad)
    print("工具箱 v" .. VERSION .. " —— 工具目录 " .. TOOL_DIR)
    print("")
    if #tools == 0 then
        print("（没有工具）")
    else
        local lastGroup = nil
        for _, t in ipairs(tools) do
            if t.group ~= lastGroup then
                lastGroup = t.group
                print("[" .. t.group .. "]")
            end
            print(string.format("  %-16s %s", t.id, t.name))
            if t.desc then
                print("      " .. t.desc)
            end
        end
    end
    if #bad and #bad > 0 then
        print("")
        print("以下工具加载失败：")
        for _, b in ipairs(bad) do
            print("  " .. b.file .. " —— " .. b.err)
        end
    end
end
local argv = { ... }
local cmd = argv[1]
local booted, nBoot = bootstrapTools()
local tools, bad, dirErr = loadTools()
if booted and nBoot > 0 then
    print(string.format("首次运行：已自动补齐 %d 个工具文件到 %s", nBoot, TOOL_DIR))
end
if cmd == "diag" then
    runDiag()
    return
end
if cmd == "about" or cmd == "version" or cmd == "-v" then
    showAbout(tools, bad)
    return
end
if cmd == "list" or cmd == "-l" then
    printList(tools, bad)
    return
end
if cmd and cmd ~= "" then
    local target = nil
    for _, t in ipairs(tools) do
        if t.id == cmd then target = t break end
    end
    if not target then
        print("没有这个工具：" .. cmd)
        print("")
        printList(tools, bad)
        return
    end
    local args = {}
    for i2 = 2, #argv do args[#args + 1] = argv[i2] end
    local ctx = cliCtx()
    local ok, err = pcall(target.run, ctx, args)
    if not ok then
        print("")
        print("!! 工具出错：" .. tostring(err))
    end
    return
end

    print("       toolbox list          列出工具")
    print("       toolbox <工具id> [参数] 直接运行")

end
